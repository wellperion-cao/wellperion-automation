📄GAS member_calendar 코드 (Survey.js 추가분)
/**
 * member_calendar — 26년 신규문의 시트에서 예약 행 추출
 * 액션: ?action=member_calendar&month=YYYY-MM (month 생략 시 당월)
 * PII: 이름 마스킹(성+**), 전화 미포함
 * 캐시: 300초
 *
 * 추가 위치: 기존 Survey.js doGet() switch(action) 블록 내
 * case 'member_calendar':
 *   return ContentService
 *     .createTextOutput(JSON.stringify(getMemberCalendar(params.month)))
 *     .setMimeType(ContentService.MimeType.JSON);
 */
function getMemberCalendar(monthParam) {
  var cache = CacheService.getScriptCache();
  var cacheKey = 'member_cal_' + (monthParam || 'current');
  var cached = cache.get(cacheKey);
  if (cached) return JSON.parse(cached);

  var ss = SpreadsheetApp.openById('12AWcAlg______GnTk'); // 26년 신규문의 스프레드시트 ID
  var sheet = ss.getSheetByName('26년 신규문의') || ss.getSheets()[0];
  var data = sheet.getDataRange().getValues();
  var headers = data[0];

  // 컬럼 인덱스 찾기
  function colIdx(name) {
    for (var i = 0; i < headers.length; i++) {
      if (String(headers[i]).trim() === name) return i;
    }
    return -1;
  }
  var iName     = colIdx('이름');
  var iProgram  = colIdx('관심 있는 프로그램 종류');
  var iStatus   = colIdx('진행상황');
  var iTs       = colIdx('타임스탬프');
  var iTourDate = colIdx('시설견학 및 상담 일정 (날짜 기록)');
  var iExp1Date = colIdx('체험1 확정시간');
  var iExp2Date = colIdx('시설 체험 일정2(날짜 기록)');

  // 대상 월 결정
  var now = new Date();
  var targetYear = now.getFullYear();
  var targetMonth = now.getMonth() + 1;
  if (monthParam) {
    var parts = String(monthParam).split('-');
    if (parts.length === 2) {
      targetYear = parseInt(parts[0], 10);
      targetMonth = parseInt(parts[1], 10);
    }
  }

  function _parseAnyDate_(val) {
    if (!val) return null;
    var s = String(val).trim();
    // 한글 날짜: 2026년 6월 5일
    var m = s.match(/(\d{4})년\s*(\d{1,2})월\s*(\d{1,2})일/);
    if (m) return new Date(parseInt(m[1]),parseInt(m[2])-1,parseInt(m[3]));
    // ISO: 2026-06-05
    m = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
    if (m) return new Date(parseInt(m[1]),parseInt(m[2])-1,parseInt(m[3]));
    // 스프레드시트 Date 객체
    if (val instanceof Date && !isNaN(val.getTime())) return val;
    return null;
  }

  function maskName(name) {
    var s = String(name || '').trim();
    if (!s) return '';
    return s.charAt(0) + '**';
  }

  function toISO(d) {
    if (!d) return null;
    var yy = d.getFullYear();
    var mm = String(d.getMonth()+1).padStart(2,'0');
    var dd = String(d.getDate()).padStart(2,'0');
    return yy + '-' + mm + '-' + dd;
  }

  function inTargetMonth(d) {
    if (!d) return false;
    return d.getFullYear() === targetYear && (d.getMonth()+1) === targetMonth;
  }

  var events = [];
  var rows = [];

  for (var r = 1; r < data.length; r++) {
    var row = data[r];
    var name = maskName(iName >= 0 ? row[iName] : '');
    var program = iProgram >= 0 ? String(row[iProgram] || '') : '';
    var status = iStatus >= 0 ? String(row[iStatus] || '') : '';
    var ts = iTs >= 0 ? _parseAnyDate_(row[iTs]) : null;

    // 견학/상담
    var tourDate = iTourDate >= 0 ? _parseAnyDate_(row[iTourDate]) : null;
    if (tourDate && inTargetMonth(tourDate)) {
      events.push({date:toISO(tourDate),kind:'견학·상담',name:name,program:program,status:status});
    }
    // 체험1
    var exp1Date = iExp1Date >= 0 ? _parseAnyDate_(row[iExp1Date]) : null;
    if (exp1Date && inTargetMonth(exp1Date)) {
      events.push({date:toISO(exp1Date),kind:'체험',name:name,program:program,status:status});
    }
    // 체험2
    var exp2Date = iExp2Date >= 0 ? _parseAnyDate_(row[iExp2Date]) : null;
    if (exp2Date && inTargetMonth(exp2Date)) {
      events.push({date:toISO(exp2Date),kind:'체험',name:name,program:program,status:status});
    }

    // 목록용 행 (전 월 포함)
    rows.push({
      name:name, program:program, status:status,
      tourDate:toISO(tourDate), exp1Date:toISO(exp1Date), exp2Date:toISO(exp2Date),
      timestamp:toISO(ts)
    });
  }

  var result = {ok:true, month:targetYear+'-'+String(targetMonth).padStart(2,'0'),
    events:events, rows:rows};
  cache.put(cacheKey, JSON.stringify(result), 300);
  return result;
}
