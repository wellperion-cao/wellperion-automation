function _inqHeaders_(sh) {
  var last = sh.getLastColumn();
  if (last < 1) return [];
  return sh.getRange(1, 1, 1, last).getValues()[0].map(function(v){ return String(v).trim(); });
}

function _inqColIdx_(headers, names) {
  var arr = Array.isArray(names) ? names : [names];
  // 1차: 정확히 일치
  for (var i = 0; i < arr.length; i++) {
    var idx = headers.indexOf(arr[i]);
    if (idx >= 0) return idx;
  }
  // 2차: 부분 일치(헤더가 키를 포함) — 실제 시트 헤더가 더 길어도 대응(예: '핸드폰 연락처'⊃'연락처', '접수 담당자'⊃'담당')
  for (var k = 0; k < arr.length; k++) {
    for (var j = 0; j < headers.length; j++) {
      if (headers[j].indexOf(arr[k]) >= 0) return j;
    }
  }
  return -1;
}

function _maskName_(n) {
  var s = String(n || '').trim();
  return s ? s.charAt(0) + '**' : '';
}

function _maskPhone_(p) {
  var s = String(p || '').trim().replace(/[^0-9]/g, '');
  if (s.length === 11) return s.slice(0,3) + '-****-' + s.slice(7);
  if (s.length === 10) return s.slice(0,3) + '-***-' + s.slice(6);
  return s ? '****' : '';
}

function _inqToISO_(val) {
  if (!val) return '';
  if (val instanceof Date && !isNaN(val.getTime())) {
    var yy = val.getFullYear(), mm = String(val.getMonth()+1).padStart(2,'0'), dd = String(val.getDate()).padStart(2,'0');
    return yy + '-' + mm + '-' + dd;
  }
  var s = String(val).trim();
  var m = s.match(/(\d{4})[\.\-\/]?\s*(\d{1,2})[\.\-\/]?\s*(\d{1,2})/);
  if (m) return m[1] + '-' + ('0'+m[2]).slice(-2) + '-' + ('0'+m[3]).slice(-2);
  return s;
}
