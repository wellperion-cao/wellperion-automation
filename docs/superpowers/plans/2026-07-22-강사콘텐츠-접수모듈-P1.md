# 강사 콘텐츠 접수 모듈 P1 (접수 폼+저장+알림) 구현 계획

> **For agentic workers:** 이 계획은 P1(MVP)만 다룬다. P2(반자동 시안)·P3(현황 패널)는 별도 계획. 각 Task는 독립 검증 가능한 산출물로 끝난다.

**Goal:** 강사가 M1 페이지에서 사진·영상·소개 텍스트를 직접 제출하면 Google Drive에 저장되고 시트에 기록되며 텔레그램 접수 알림이 오는 폼을 만든다(→ 완료 즉시 부서장 카톡방 공지·수집 시작).

**Architecture:** 신규 전용 GAS 웹앱(강사접수)이 doPost로 메타=시트 appendRow + 파일=base64→DriveApp.createFile 저장 후 _notifyTelegram. 프론트=독립 폼 HTML(다중 파일 base64 POST). M1 페이지(`wellperion_guide(main).html` id="M1" article)에 `m1-instructor-intake` 섹션 신설해 iframe 임베드.

**Tech Stack:** Google Apps Script(웹앱·DriveApp·SpreadsheetApp·UrlFetchApp), 정적 HTML/JS(FileReader base64), GitHub Pages.

## Global Constraints
- 브랜드 금지어: "피트니스"→"스포츠클럽" (폼 안내문에도 적용).
- 영상 파일 상한 = 50MB(base64 ~33% 팽창·GAS 페이로드 한계). 초과 시 "드라이브 링크 붙여넣기" 폴백 필드로 안내.
- 강사 제출 = 공식계정 게시 동의 전제(동의 체크박스 필수). 미승인 상태에선 발행 안 됨(검수 게이트는 P2).
- GAS는 pytest 불가 — 검증은 `clasp` 배포 후 curl/브라우저 실측(POST 응답 + 시트 행 + Drive 파일 + 텔레그램 수신).
- 앵커 id 보존 규율(가이드 SPA): 기존 `m1-*` 앵커 유지, 신설만 추가.
- 파일 업로드 MIME 화이트리스트(image/jpeg,image/png,image/webp,video/mp4) — 조달 `putPhoto` L95 패턴 준수(주입 차단).

```javascript
var ALLOW_MIME = {'image/jpeg':1,'image/png':1,'image/webp':1,'video/mp4':1};
function _saveFile_(b64, mime, fname, folder){
  if(!ALLOW_MIME[mime]) throw new Error('허용되지 않은 형식: '+mime);
  var raw = b64.indexOf(',')>=0 ? b64.split(',')[1] : b64;   // data: 접두사 방어(VOC L462)
  var blob = Utilities.newBlob(Utilities.base64Decode(raw), mime, fname);
  var f = folder.createFile(blob);
  f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return f.getUrl();
}
```

- [ ] **Step 3: doPost 작성**
```javascript
function doPost(e){
  try{
    var d = JSON.parse(e.postData.contents);
    if(!d.name || !d.team || !d.agree) return _json({ok:false, err:'필수 항목 누락'});
    var root = DriveApp.getFolderById(PropertiesService.getScriptProperties().getProperty('INTAKE_DRIVE_FOLDER_ID'));
    var folder = root.createFolder(d.team+'_'+d.name+'_'+_stamp());
    var urls = (d.photos||[]).map(function(p){ return _saveFile_(p.b64,p.mime,p.fname,folder); });
    var vurl = d.video ? _saveFile_(d.video.b64,d.video.mime,d.video.fname,folder) : (d.videoLink||'');
    var sh = SpreadsheetApp.openById(PropertiesService.getScriptProperties().getProperty('INTAKE_SHEET_ID')).getSheets()[0];
    var row = [new Date(), d.name, d.team, d.intro||'', d.benefit||'', urls.join('\n'), vurl, folder.getUrl(), '접수'];
    sh.appendRow(row);
    _notifyTelegram('🎬 새 강사 콘텐츠 접수: '+d.name+' / '+d.team+' (사진 '+urls.length+'장'+(vurl?'·영상':'')+')');
    return _json({ok:true, drive_folder:folder.getUrl(), sheet_row:sh.getLastRow()});
  }catch(err){ return _json({ok:false, err:String(err)}); }
}
function _json(o){ return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function _stamp(){ return Utilities.formatDate(new Date(),'Asia/Seoul','yyyyMMdd_HHmmss'); }
```

- [ ] **Step 4: _notifyTelegram 이식** (Survey.js:834-839 동형)
```javascript
function _notifyTelegram(text){
  var p = PropertiesService.getScriptProperties();
  var token = p.getProperty('BOT_TOKEN'), chat = p.getProperty('INTAKE_CHAT_ID');
  if(!token||!chat) return;
  UrlFetchApp.fetch('https://api.telegram.org/bot'+token+'/sendMessage',
    {method:'post', payload:{chat_id:chat, text:text}, muteHttpExceptions:true});
}
```

```javascript
function toB64(file){ return new Promise(function(res){ var r=new FileReader(); r.onload=function(){res({b64:r.result,mime:file.type,fname:file.name});}; r.readAsDataURL(file); }); }
async function submit(){
  var photos = await Promise.all([...photoInput.files].slice(0,5).map(toB64));
  var video = videoInput.files[0] && videoInput.files[0].size <= 50*1024*1024 ? await toB64(videoInput.files[0]) : null;
  if(videoInput.files[0] && !video){ alert('영상이 50MB를 넘어요. 드라이브 링크로 넣어주세요.'); return; }
  var body = {name:nameEl.value, team:teamEl.value, intro:introEl.value, benefit:benefitEl.value,
             agree:agreeEl.checked, photos:photos, video:video, videoLink:videoLinkEl.value};
  var r = await fetch(GAS_PROD, {method:'POST', body:JSON.stringify(body)}); // no-cors 주의: text/plain content-type로 preflight 회피(Survey.js 패턴)
  var j = await r.json();
  msgEl.textContent = j.ok ? '접수 완료! 감사합니다 🙏' : ('오류: '+j.err);
}
```
